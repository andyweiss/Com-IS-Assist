//Deklaration der alloc function mit sz als Variable. Wenn diese ausgeführt wird, wird ein Teil des Buffers für eine bestimmte Variable reserviert.
function alloc(sz) global()//weist memory einer globalen var zu, sz ist die größe des Speichers in bytes
(

// sinc filter for true peak
function sinc_gen_val() global(srate sinc_sz) local(sincpos windowpos) instance(slice_pos) (
  windowpos = (2.0 * $pi / sinc_sz) * slice_pos;
  sincpos = $pi * (slice_pos - sinc_sz * .5);
  slice_pos += 1;
  (0.53836 - cos(windowpos)*0.46164) * sin(sincpos) / sincpos;
);

function sinc_gen_slice(cs, o*) instance(sinc_gen_val slice_pos) local(x) global(sinc_sz) (
  slice_pos = cs;
<? 
  use_mem_sinc ? (
    printf("o.v = alloc(sinc_sz);  loop(x=0;sinc_sz, o.v[x] = sinc_gen_val(); x+=1); ");
  ) : (
    loop(x=0;sinc_sz, printf("o.v%02d = sinc_gen_val();%s",x,(x&3)==3?"\n": " "); x += 1)
  );
?>
);

function sinc_init() global(srate) instance(sinc_gen_slice) (
  sinc_gen_slice(srate < 96000 ? .25 : .5,this.s1);
  srate < 96000 ? (
    sinc_gen_slice(.5, this.s2);
    sinc_gen_slice(.75, this.s3);
  );
);

//Proc Funktion
function proc(lspl)
  local(pspl f1p0 f2p0) 
  instance(wt ch_rms_sum ch_lufs_sum pkval hipkval clips
           f1p1 f1p2 f2p1 f2p2)
  global(f1a1 f1a2 f1b0 f1b1 f1b2
         f2a1 f2a2 f2b0 f2b1 f2b2
         srate cfg_peak
         lval rval win_pos global_peak sinc_sz
         sinc.s1* sinc.s2* sinc.s3*
         )
(
  cfg_peak == 1 || cfg_peak == 2 ? (
    <?
      use_mem_sinc ? (
        printf("mem_insert_shuffle(this.h, sinc_sz, lspl);");
      ) : (
        loop(x=sinc_sz-1;sinc_sz-1, printf("this.h%02d = this.h%02d;%s",x,x-1, (x&3)==0 ? "\n" : " "); x-= 1; );
        printf(" this.h00 = lspl;");
      );
    ?>
    pspl = max(
        abs(this.h<? printf(use_mem_sinc ? "[%d]" : "%02d",sinc_sz/2) ?> ),
        <? emit_sinc("this","sinc.s1") ?>
       );
    srate < 96000 ? pspl = max(pspl,max( <? emit_sinc("this","sinc.s2") ?>,  <? emit_sinc("this","sinc.s3") ?>));
  ) : 
  cfg_peak == 3 || cfg_peak == 4 ? (
    pspl=abs(lspl);
  );
  cfg_peak ? (
    pspl > pkval ? (
      pkval=pspl;
      pspl > hipkval ? (
        hipkval=pspl;
        global_peak = max(global_peak,pspl);
      );
    );
    pspl > 1 ? clips += 1;
  );

  win_pos == 0 ? ch_rms_sum=ch_lufs_sum=0;

  rval += (ch_rms_sum += lspl*lspl);
  
  lspl *= wt;

  //Anwendung der LUFS Filter 1
  f1p0=lspl-f1a1*f1p1-f1a2*f1p2;
  lspl=f1b0*f1p0+f1b1*f1p1+f1b2*f1p2;
  f1p2=f1p1;
  f1p1=f1p0;
  
  //Und Filter 2
  f2p0=lspl-f2a1*f2p1-f2a2*f2p2;
  lspl=f2b0*f2p0+f2b1*f2p1+f2b2*f2p2;
  f2p2=f2p1;
  f2p1=f2p0;
  
  lval += (ch_lufs_sum += lspl*lspl);
);

  local(pCspl Cf1p0 Cf2p0)
  instance(wt ch_Crms_sum ch_Clufs_sum pkCval hipkCval clips
          Cf1p1 Cf1p2 Cf2p1 Cf2p2)
  global(f1a1 f1a2 f1b0 f1b1 f1b2
         f2a1 f2a2 f2b0 f2b1 f2b2
         srate cfg_peak
         lCval rCval win_pos global_Cpeak sinc_sz
         sinc.s1* sinc.s2* sinc.s3*
         )(
  win_pos == 0 ? ch_Crms_sum=ch_Clufs_sum=0;

  rCval += (ch_Crms_sum += Clspl*Clspl);
  
  Clspl *= sqrt(2);

  //Anwendung der LUFS Filter 1
  Cf1p0=Clspl-f1a1*Cf1p1-f1a2*Cf1p2;
  Clspl=f1b0*Cf1p0+f1b1*Cf1p1+f1b2*Cf1p2;
  Cf1p2=Cf1p1;
  Cf1p1=Cf1p0;
  
  //Und Filter 2
  Cf2p0=Clspl-f2a1*Cf2p1-f2a2*Cf2p2;
  Clspl=f2b0*Cf2p0+f2b1*Cf2p1+f2b2*Cf2p2;
  Cf2p2=Cf2p1;
  Cf2p1=Cf2p0;
  
  lCval += (ch_Clufs_sum += Clspl*Clspl);
);

//Bandpass LUFS_Filter
function init_lufs_filters()
  local(Vh Vb db f0 Q K a0)
  global(f1a1 f1a2 f1b0 f1b1 f1b2
         f2a1 f2a2 f2b0 f2b1 f2b2 srate)
(
  // f1,f2 could be combined into a 5th order filter
//Shelving 2. Ordnung---------------------------------------------------
//Hier werden die Filterkoeffizienten abhaengig von der Sample-Rate berechnet
  db=3.999843853973347; //Filter Gain
  f0=1681.974450955533; //base frequency
  Q=0.7071752369554196; //Filter-Güte
  K=tan($pi*f0/srate); //Tangens der halben Filterbandbreite in Bezug auf die Abtastrate des Signals (Pi*Frequenz)/sr ca. 0.11
  Vh=pow(10, db/20); 
  Vb=pow(Vh, 0.4996667741545416);
  
  //Folgende sind alles Filterkoeffizienten, werden aus db f0 Q und SR berechnet
  a0=1+K/Q+K*K; // K als Variable, alles ca. 0
  f1a1=2*(K*K-1)/a0; // -1.69, K und a0 als Variablen           -----alpha1
  f1a2=(1-K/Q+K*K)/a0; // 0.73, K und a0 als Variablen          -----alpha2
  f1b0=(Vh+Vb*K/Q+K*K)/a0; //1.535 Vh Vb K und a0 als Variablen -----beta0
  f1b1=2*(K*K-Vh)/a0; //                                        -----beta1
  f1b2=(Vh-Vb*K/Q+K*K)/a0; //                                   -----beta2

//Hochpass 2. Ordnung---------------------------------------------------
  f0=38.13547087602444;
  Q=0.5003270373238773;
  K=tan($pi*f0/srate);
  f2a1=2*(K*K-1)/(1+K/Q+K*K);   //-----alpha1
  f2a2=(1-K/Q+K*K)/(1+K/Q+K*K); //-----alpha2
  f2b0=1;                       //-----beta0
  f2b1=-2;                      //-----beta1
  f2b2=1;                       //-----beta2
);

//Correlation Meter--------------------------------------------
 signalThreshold = 0.0001;
 
 // Sample memory
 history    = 2000;
 historyS   = 2000; //Modifizierung durch Autor
 
 // Sample history statistics and properties
 historyLength = 45; // Default: ~1ms Window at 44.1 kHz
 historyDouble = 90; // 2x historyLength, used by methods
 historyScaler = 0.02222222222; // For mean calculation
 
 // Correlation result related values
 correlation = 0.0; // Raw Pearson result
 correlationS = 0.0; //Modifizierung durch Autor
 indicatorPosition = 0.0; // Smoothed position
 indicatorPositionS = 0.0; //Modifizierung durch Autor
 
 // Used for level detection
 absMeanL = 0.0;
 absMeanR = 0.0;
 absMeanLs = 0.0; //Modifizierung durch Autor
 absMeanRs = 0.0; //Modifizierung durch Autor
 
 // One-pole smooth movement filter coeffs
 slowDownA = 1.0;
 slowDownB = 0.0;
  
  // Fast replacement for (1.0 / value) expressions
  //
  function fastReciprocal (value) local (factor)
  (
    factor = invsqrt(historyLength);
    (factor * factor);
  );
  
   //Modifizierung durch Autor
  function fastReciprocalS (valueS) local (factorS)
  (
    factorS = invsqrt(historyLength);
    (factorS * factorS);
  );
  
  // One-pole LP filter to, well, slow down the
  // indicator's movement.
  //
  function slowDown (value) local (state)
  (
    state = (value * slowDownA) + (state * slowDownB);
    state;
  );
  
   //Modifizierung durch Autor
  function slowDownS (valueS) local (stateS)
  (
    stateS = (valueS * slowDownA) + (stateS * slowDownB);
    stateS;
  );
  
  // The correlation detection window will be 1 ms
  // in any project with any samplerate.
  //
  function resizeHistory () local (samplesPerMS)
  (
    samplesPerMS = ceil(srate * 0.001); // round up +1 for safety

    (historyLength != samplesPerMS) ? 
    (
      historyLength = samplesPerMS;
      historyDouble = historyLength * 2;
      historyScaler = fastReciprocal(historyLength); // 1.0/historyLength
      
      memset(8192+history, 0.0, historyDouble);
    );
  );
  
  
   //Modifizierung durch Autor
  function resizeHistoryS () local (samplesPerMS)
  (
    samplesPerMS = ceil(srate * 0.001); // round up +1 for safety
    
    (historyLength != samplesPerMS) ? 
    (
      historyLength = samplesPerMS;
      historyDouble = historyLength * 2;
      historyScaler = fastReciprocalS(historyLength); // 1.0/historyLength
      
      memset(8192+history+historyS, 0.0, historyDouble);
    );
  );
  
  // Move everything in the history back one sample,
  // then import the latest incoming samples.
  //
  //Modifizierung durch Autor
  function shuffle (newSampleL, newSampleR)
  (
    memcpy(8192+history, 8192+history+1, historyDouble);
    
    history[8192+historyLength-1] = newSampleL;
    history[8192+historyDouble-1] = newSampleR;
  );

  function pearsonProduct () local (position, sumL, sumR, meanL, meanR, devL, devR, sqrMeanL, sqrMeanR, sumAbove, numBelow)
  (
    rounds += 1;
    rounds %= historyLength;
    
    // Every Millisecond
    (rounds == 0) ? 
    (
      // Just in case the sample rate's changed
      resizeHistory();
      
      // SUM
      sumL  = 0;
      sumR  = 0;
      
      position = 0;
      while (position < historyLength)
      ( //Modifizierung durch Autor
        sumL += history[8192+position];
        sumR += history[8192+position+historyLength];
        
        position += 1;
      );
      
      // MEAN
      meanL = (sumL * historyScaler);
      meanR = (sumR * historyScaler);
      
      absMeanL = abs(meanL);
      absMeanr = abs(meanR);
      
      // DEVIATION
      sqrMeanL = 0;
      sqrMeanR = 0;
      
      position = 0;
      while (position < historyLength)
      ( //Modifizierung durch Autor
        sqrMeanL += sqr(history[8192+position] - meanL);
        sqrMeanR += sqr(history[8192+position+historyLength] - meanR);
        
        position += 1;
      );
      
      devL = sqrt(sqrMeanL * historyScaler);
      devR = sqrt(sqrMeanR * historyScaler);
      
      // SUM ABOVE
      sumAbove = 0.0;
      
      position = 0;
      while (position < historyLength)
      ( //Modifizierung durch Autor
        sumAbove += (history[8192+position] - meanL) * (history[8192+position+historyLength] - meanR);
        
        position += 1;
      );
      
      // NUM BELOW
      numBelow = historyLength * devL * devR;
      
      // CORRELATION RESULT
      //
      // The value this entire processor is built around
      //
      correlation = sumAbove / numBelow;
    );
  );

  function pearsonProductS () local (positionS, sumLs, sumRs, meanLs, meanRs, devLs, devRs, sqrMeanLs, sqrMeanRs, sumAboveS, numBelowS)
  (
    roundsS += 1;
    roundsS %= historyLength;
    
    // Every Millisecond
    (roundsS == 0) ? 
    (
      // Just in case the sample rate's changed
      resizeHistoryS();
      
      // SUM
      sumLs  = 0;
      sumRs  = 0;
      
      positionS = 0;
      while (positionS < historyLength)
      (
        sumLs += historyS[8192+history+positionS];
        sumRs += historyS[8192+history+positionS+historyLength];
        
        positionS += 1;
      );
      
      // MEAN
      meanLs = (sumLs * historyScaler);
      meanRs = (sumRs * historyScaler);
      
      absMeanLs = abs(meanLs);
      absMeanRs = abs(meanRs);
      
      // DEVIATION
      sqrMeanLs = 0;
      sqrMeanRs = 0;
      
      positionS = 0;
      while (positionS < historyLength)
      (
        sqrMeanLs += sqr(historyS[8192+history+positionS] - meanLs);
        sqrMeanRs += sqr(historyS[8192+history+positionS+historyLength] - meanRs);
        
        positionS += 1;
      );
      
      devLs = sqrt(sqrMeanLs * historyScaler);
      devRs = sqrt(sqrMeanRs * historyScaler);
      
      // SUM ABOVE
      sumAboveS = 0.0;
      
      positionS = 0;
      while (positionS < historyLength)
      (
        sumAboveS += (historyS[8192+history+positionS] - meanLs) * (historyS[8192+history+positionS+historyLength] - meanRs);
        
        positionS += 1;
      );
      
      // NUM BELOW
      numBelowS = historyLength * devLs * devRs;
      
      // CORRELATION RESULT
      //
      // The value this entire processor is built around
      //
      correlationS = sumAboveS / numBelowS;
    );
  );

//-------------------------------------------------------------------------------------------dB Formatierung/Skalierung
function format_dbv(hv)
(
  hv <= -72 ? "" : sprintf(#, "%+.1f", hv); 
);
function format_db(v)
(
  v <= 10^(-72/20) ? "" : format_dbv(log(v)*20/log(10));
);
function scale_db1(v) global(cfg_yscale)
(
  v <= -72 ? 0 : v >= 0 ? 1 : pow(v*1/72+1,cfg_yscale);
);
function scale_v(v) global(cfg_yscale)
(
  v <= 10^(-72/20) ? 0 : v >= 1 ? 1 : pow(log(v)*20/log(10)/72+1,cfg_yscale);
);
